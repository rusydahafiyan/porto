PostgreSQL Schema & TypeScript Pricing Engine

-- 1. COMPLETE POSTGRESQL SCHEMA (SUPABASE + RLS + ESCROW + CONTEXTUAL CHAT)
CREATE TYPE engine_type AS ENUM ('DOMESTIC_MALL', 'INTL_FLIGHT');
CREATE TYPE discount_mode AS ENUM ('BUYER_LOYALTY', 'SPLIT_BENEFIT', 'MARGIN_HUNTER');
CREATE TYPE order_state AS ENUM (
  'DP_LOCKED_SAFE_TO_BUY', 'IN_STORE_HUNTING', 'SUBSTITUTION_PENDING',
  'PURCHASED_RECEIPT_VERIFIED', 'IN_TRANSIT', 'DELIVERED_24H_SLA',
  'SETTLED_RELEASED', 'HIT_AND_RUN_FORFEITED', 'DISPUTED_UNBOXING_REVIEW'
);
CREATE TYPE box_tolerance AS ENUM ('MINT_ONLY', 'MINOR_DENT_OK', 'NO_BOX_SAVE_SPACE');

CREATE TABLE users (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  full_name VARCHAR(120) NOT NULL,
  masked_phone_proxy VARCHAR(32) UNIQUE NOT NULL,
  kyc_passport_verified BOOLEAN DEFAULT FALSE,
  escrow_wallet_idr BIGINT DEFAULT 0,
  created_at TIMESTAMPTZ DEFAULT NOW()
);

CREATE TABLE trips (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  jastipper_id UUID REFERENCES users(id) ON DELETE CASCADE,
  engine engine_type NOT NULL,
  origin_city VARCHAR(80) NOT NULL,
  destination_city VARCHAR(80) NOT NULL,
  flight_code VARCHAR(24),
  baggage_max_kg NUMERIC(5,2) DEFAULT 30.00,
  baggage_used_kg NUMERIC(5,2) DEFAULT 0.00,
  customs_fob_usd_accumulated NUMERIC(8,2) DEFAULT 0.00, -- PMK 203 USD 500 cap
  bpom_skincare_units INT DEFAULT 0 CHECK (bpom_skincare_units <= 5),
  created_at TIMESTAMPTZ DEFAULT NOW()
);

CREATE TABLE orders (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  trip_id UUID REFERENCES trips(id),
  buyer_id UUID REFERENCES users(id),
  seller_id UUID REFERENCES users(id),
  state order_state NOT NULL DEFAULT 'DP_LOCKED_SAFE_TO_BUY',
  discount_distribution discount_mode NOT NULL DEFAULT 'SPLIT_BENEFIT',
  is_concert_ticket BOOLEAN DEFAULT FALSE,
  event_date TIMESTAMPTZ, -- H+1 Escrow Lock rule
  dp_locked_idr BIGINT NOT NULL,
  total_payable_idr BIGINT NOT NULL,
  thermal_receipt_sha256 TEXT,
  created_at TIMESTAMPTZ DEFAULT NOW()
);

CREATE TABLE order_items (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  order_id UUID REFERENCES orders(id) ON DELETE CASCADE,
  sku_barcode VARCHAR(64),
  title TEXT NOT NULL,
  tag_currency CHAR(3) NOT NULL,
  tag_price NUMERIC(12,2) NOT NULL,
  fx_buffer_pct NUMERIC(4,2) DEFAULT 3.00,
  volumetric_weight_kg NUMERIC(6,2),
  shoe_box_pref box_tolerance DEFAULT 'MINOR_DENT_OK',
  anti_theft_tag_removed BOOLEAN DEFAULT FALSE
);

CREATE TABLE contextual_messages (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  order_id UUID REFERENCES orders(id) ON DELETE CASCADE,
  sender_id UUID REFERENCES users(id),
  card_type VARCHAR(32) DEFAULT 'TEXT', -- TEXT, SUBSTITUTION_3M, PRICE_DELTA, RECEIPT_OCR
  sanitized_body TEXT NOT NULL,
  pii_redacted BOOLEAN DEFAULT FALSE,
  expires_at TIMESTAMPTZ, -- 3-minute countdown SLA for substitutions
  created_at TIMESTAMPTZ DEFAULT NOW()
);

CREATE TABLE escrow_transactions (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  order_id UUID REFERENCES orders(id),
  pjp_provider VARCHAR(24) DEFAULT 'XENDIT_SPLIT',
  phase VARCHAR(16) NOT NULL, -- PHASE_1_DP, PHASE_2_FINAL, FORFEIT_COMPENSATION
  amount_idr BIGINT NOT NULL,
  status VARCHAR(24) NOT NULL,
  created_at TIMESTAMPTZ DEFAULT NOW()
);

CREATE TABLE dispute_logs (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  order_id UUID REFERENCES orders(id),
  uncut_unboxing_video_url TEXT NOT NULL,
  resolution VARCHAR(32) DEFAULT 'UNDER_REVIEW',
  created_at TIMESTAMPTZ DEFAULT NOW()
);
